\begin{frame}{研究目的}

    非易失性内存的特性有别于传统存储设备，为编程带来新的挑战

    \vspace{1em}

    \begin{itemize}[<+->]
        \item NVM具有非易失性
        \begin{itemize}
            \item 程序的执行效果能够被保留到系统崩溃后……
            \item ……但这也包括程序中已经发生的错误 
        \end{itemize}
        \item 对内存的写操作是异步生效的
        \begin{itemize}
            \item 程序的执行效果可能只保留了一部分……
            \item ……并且它们是乱序写回的
        \end{itemize}
        \item 系统崩溃在不同场景下的作用范围不同
        \begin{itemize}
            \item 单机场景：完全崩溃模型
            \item 分布式场景：部分崩溃模型
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{正确性条件}

    无论采取何种语义和验证方式，NVM正确性保障的目标相同，即证明其\highlight{崩溃一致性}：
    \vfill
    \begin{block}{崩溃一致性}
        称运行于NVM的程序是崩溃一致的，当且仅当在其执行的任意时刻，其非易失性内存状态都满足给定的安全属性$\phi$。
    \end{block}
    \vfill
    直观理解：无论如何崩溃，程序恢复后的状态必然是一致的。
\end{frame}

\begin{frame}{声明语义}
    执行图中的节点是程序的访存操作，有向边则是操作间的序关系
    \vfill
    \centering
    \begin{tikzpicture}[executiongraph]
        \node[visible] (W_a) {$\writeE(a, 1)$};
        \node[visible] (W_b) [right=3.2cm of W_a] {$\writeE(b, 1)$};

        \draw[solidarrow] (W_a) -- node[above] {\po} (W_b);
        \draw[->, dotted, color=Sepia, line width=1pt, shorten >=2pt, shorten <=2pt, bend left=35]
            (W_b) to node[below] {\nvo} (W_a);
    \end{tikzpicture}

    \vfill
    \begin{itemize}
        \item 程序顺序$\po$刻画了 $a \assigns 1$ 先于 $b \assigns 1$ 执行
        \item 持久化顺序$\nvo$刻画了 $b \assigns 1$ 先于 $a \assigns 1$ 写回主存
    \end{itemize}
\end{frame}

\begin{frame}{公理}
	执行图中的序关系并非任意，而是必须符合内存模型特定的公理

\begin{frame}{总体思路}
    检查鲁棒性要求遍历程序状态，并依次检查其可达性。例如，对于程序$a \assigns 1 \parallel b \assigns 1$

    \vfill
    \centering
    \begin{tabular}{ccc}
    \toprule
      程序状态 & 是否可达 & 是否无崩溃可达 \\

    \bottomrule
    \end{tabular}

	\begin{block}{核心问题}
		\begin{center}
            给定状态$s$，如何检查$s$是否可达？
		\end{center}
	\end{block}

\end{frame}

\begin{frame}{恢复观测线程}
    由数学归纳法，只需检查给定状态是否在一次崩溃后可达。
    \vfill
    恢复观测线程是在原本程序之外的一个\highlight{虚拟线程}，包含对所有内存位置的一次读取，其可用于将状态嵌入到崩溃前的执行图中。

    \begin{columns}
    	\begin{column}{0.4\textwidth}
    		\begin{concurrentcode}
    			x \assigns 1; & y \assigns 2;  \\
    			\FLUSH x;     & \FLUSH y;       \\
    			a \assigns y; & b \assigns x;  \\
    			x \assigns a; & y \assigns b;  \\
    		\end{concurrentcode}
    	\end{column}
    
    	\begin{column}{0.5\textwidth}
    		\begin{figure}
    			\includegraphics[width=\linewidth]{pic/reduction-exp.pdf}
    		\end{figure}
    	\end{column}
    \end{columns}
\end{frame}

\begin{frame}{DTPO约束}
    除已有的NVM的公理，使用恢复观测线程得到的执行图需额外满足一定条件。为此定义关于刷新指令的DTPO约束：
    \[
    G.\dtpo \triangleq \bigcup_{x \in \variables} \flushE_x \times \left\{ w \in \writeE_x \mid \exists w' \in \domain(G.\rrf), (w', w) \in G.\mo  \right\}
    \]
    \[
    G.\dtpo; \tpo~\text{是非自反的}
    \]

    \begin{block}{直观}
        刷新指令的效果是缓存中的值立刻写回主存。因此，假设恢复观测线程读到$w'$，则$w'$后发生的写事件$w$都一定发生在刷新指令之后，否则与恢复后观测到$w'$的事实矛盾。
    \end{block}
\end{frame}

\begin{frame}{可达性检查}
    \begin{block}{核心问题}
		\begin{center}
            给定状态$s$，如何检查$s$是否可达？
		\end{center}
	\end{block}

    \begin{itemize}
        \item 为程序插桩，引入恢复观测线程
        \item 得到插桩后的执行图
        \item 检查执行图是否满足公理以及DTPO约束
    \end{itemize}
\end{frame}

\begin{frame}{状态遍历}
    将约束编码为 SMT 公式，借助 DPLL 算法可高效遍历状态空间。

\begin{columns}
	\begin{column}{0.5\textwidth}
		\begin{align}

			     & \land x_1 = 1 \land a = y_1 \land x_2 = a \tag{线程1}  \\
			     & \land y_2 = 2 \land b = x_3 \land y_3 = b \tag{线程2} \\
			     & \land r_1 = x_4 \land r_2 = y_4 \tag{恢复观测线程}
		\end{align}
	\end{column}

\end{frame}

\section{基于不变式综合的验证方法}

\begin{frame}{程序逻辑}
    另一种常见的验证方法是基于程序逻辑生成验证条件
	\begin{itemize}
		\item 串行程序：霍尔逻辑 $\{P\}~c~\{Q\}$
		\item 多线程并发程序：Owicki-Gries 推理$R, G \vdash \{P\}~c~\{Q\}$
        \begin{equation}
			      \inferrule{
				      R_1, G_1 \text{ 与 }~R_2, G_2 \text{ 不相干 } \\\\
				      R_1, G_1 \vdash \{P_1\}~c_1~\{Q_1\} \\\\
				      R_2, G_2 \vdash \{P_2\}~c_2~\{Q_2\}
			      }{
				      \{Q\} \cup R_1 \cup R_2, G_1 \cup G_2 \vdash \{P_1 \land P_2\}~c_1~\parallel~c_2~\{Q\}
			      }
                  \tag{不相干性}
        \end{equation}
        \item 运行于NVM 的并发程序：？
	\end{itemize}
\end{frame}

\begin{frame}{一种面向NVM的程序逻辑（\S~5.3）}

    在Owicki-Gries推理的基础上可以得到一种适用于 NVM 的程序逻辑。
    \vfill
    \highlight{主要挑战}：程序逻辑无法直接对崩溃后的状态进行推理。
    \vfill
    \begin{block}{辅助变量}
        \begin{enumerate}
            \item 对每个共享变量 $x$，引入一个新的共享变量 $x'$
            \item $x$代表变量当前在缓存中的值，$x'$则代表变量在主存中的实际值。
            \item 崩溃后，从NVM恢复的值是$x'$。
        \end{enumerate}
    \end{block}
    \vfill
    可以证明引入辅助变量后的内存模型和原始的模型语义等价。
\end{frame}

\begin{frame}{稳定性}
	除引入辅助变量，证明中的所有公式还需额外满足\highlight{稳定性}约束。

	\vfill
	\begin{definition}[稳定性]
		若定义在 $\variables \cup \variables' \cup \registers$ 上的一阶逻辑公式 $\phi$ 满足
		\[
			\forall x \in \variables, \phi \rightarrow \phi[x\backslash x']
		\]
		则称 $\phi$ 是\emph{稳定的}。
		\label{def:stability}
	\end{definition}
    \vfill
    稳定性约束无法用霍尔推导规则的形式给出。其在逻辑层面描述了共享变量非确定写回副本的过程。
\end{frame}

\begin{frame}{示例}
    目标：证明$\phi \triangleq z = 1 \rightarrow x = 1$在非易失性内存状态上恒成立

    \begin{figure}
    \begin{concurrentcode}
        $\{I_1: I\}$ & $\{I_5: I \land a = 1 \rightarrow y = 1\}$ \\
        x \assigns 1; & \WHILE a $\neq$ 1 \DO \\
        $\{I_2: I \land x = 1\}$ & \qquad $\{I_6: I\}$ \\
        \FLUSH x; & \qquad a \assigns y; \\
        $\{I_3: I \land x' = 1\}$ & \qquad $\{I_7: I \land a = 1 \rightarrow y = 1\}$ \\
        y \assigns 1; & $\{I_8: I \land y = 1\}$ \\
        $\{I_4: I\}$ & z \assigns 1; \\
        & $\{I_9: I\}$ \\
    \end{concurrentcode}
    \caption*{$I \triangleq x' = 1 \rightarrow x = 1 \land z' = 1 \rightarrow z = 1 \land y = 1 \rightarrow x' = 1 \land z = 1 \rightarrow x' = 1 \land z' = 1\rightarrow x' = 1$}
    \end{figure}
    
    称$I_1, \ldots, I_9$ 为对应程序位置的\highlight{不变式}。
\end{frame}

\begin{frame}{验证条件}

    为每个程序位置$i$引入未知谓词$I_i$，代表该位置的\emph{不变式}，则程序可被编码为验证条件。公式分为以下部分：
    \vfill
    \begin{itemize}
        \item 初始值
        \item 霍尔三元组有效性
        \item 不相干性
        \item 稳定性约束
        \item 安全属性
    \end{itemize}
\end{frame}

\begin{frame}{示例}
	\begin{align*}
		\psi & \triangleq x = y = z = a = x' = y' = z' = 0 \rightarrow I_1(x, y, z, a, x', y', z') \ldots \tag{初始化}                          \\
		     & \land I_1(x, y, z, a, x', y', z') \land x' = 1 \rightarrow I_2(x', y, z, a, x', y', z') \ldots \tag{有效性}                      \\
		     & \land I_1(x, y, z, a, x', y', z') \land I_8(x, y, z, a, x', y', z') \rightarrow I_1(x, y, 1, a, x', y', z') \ldots \tag{不相干性} \\
		     & \land I_1(x, y, z, a, x', y', z') \rightarrow  I_1(x, y, z, a, x, y', z') \ldots \tag{稳定性}                                    \\
		     & \land I_1(x, y, z, a, x', y', z') \rightarrow  \phi[\overrightarrow{x'/x}] \ldots \tag{安全性}                                   \\
	\end{align*}
\end{frame}

\begin{frame}{验证方法对比}
    \begin{itemize}[<+->]
        \item 两者都通过某种方式将崩溃消除，只考虑无崩溃执行
        \item 都需要额外的约束建模崩溃的语义：DTPO约束 v.s. 稳定性约束
        \item 鲁棒性验证基于声明语义，而不变式综合基于操作语义
        % \item 声明语义要求访存操作有限，故处理无界循环时有误报
    \end{itemize}
\end{frame}
