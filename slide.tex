\begin{frame}{语义框架}
    内存模型的刻画方法主要有两种：
    \begin{itemize}
        \item 操作语义：通过\highlight{标记变迁系统}模拟实际的内存层级行为
        \begin{equation}
        \inferrule{
			B(\tau) = b \\\\ b = b_1 \cdot \writeE(x, v) \cdot b_2 \\\\
			\writeE(x, \_) \not\in b_2
		}{
			(B, C, m) \xrightarrow{\tau,\, \readE(x, v)} (B, C, m)
		} \tag{局部读取规则}
        \end{equation} 
        \item 声明语义：通过\highlight{执行图}上的公理约束访存操作的序关系
        \begin{equation}
            \forall x \in \variables. \domain([\writeE_x \cup \RMW_x \cup \readE_x \cup \RexE_x]; \tpo; [G.P_x]) \subseteq G.P \tag{持久化公理}
        \end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{声明语义（\S~6.4）}

    执行图与崩溃的交错序列构成\highlight{执行链}。
    
    \begin{itemize}
        \item 执行链中的每一张图表示一次无崩溃执行。
        \item 相邻的执行图的事件集并不独立，取决于崩溃的范围。
    \end{itemize} 

    \begin{definition}[执行图]
	\label{def:execution-graph}
	执行图 $G \in \egraph$ 是一个五元组 $(E, V, P, \rf, \mo)$，其中：
	\begin{itemize}
		\item $E \subseteq \eventset$ 是一个有限事件集合。
		\item $V$ 是\emph{可见}事件集合，满足 $E \cap (\initset \cup \readE \cup \RMW \cup \RexE \cup \fenceE) \subseteq V \subseteq E$。
		\item $P$ 是已经\emph{持久化}的事件集合，满足 $E \cap \initset_\bot \subseteq P \subseteq V \cap (\writeE \cup \RMW \cup \readE \cup \RexE)$。
		\item $\rf$ 是 $E$ 上的一个读取顺序。
		\item $\mo$ 是 $E$ 上的一个修改顺序。
	\end{itemize}
    \end{definition}
\end{frame}

\begin{frame}{公理}
    
    执行链中的每一张执行图都应当满足下述公理

    \vfill
    \centering
    \begin{tabular}{>{\raggedright\arraybackslash}p{0.7\linewidth}r}
      \hline
      $\tpo$ 在 $G.E \setminus \readE$ 上是全序 & {\textsc{(tso-total)}} \\
      \hline
      $G.\mo \subseteq \tpo$ & {\textsc{(tso-mo)}} \\
      \hline
      $G.\rf \subseteq \tpo \cup G.\po$ & {\textsc{(tso-rf)}} \\
      \hline
      $G.\po \setminus ((\writeE \cup \flushE) \times \readE) \subseteq \tpo$ & {\textsc{(tso-po)}} \\
      \hline
      $G.\fr(\tpo) \subseteq \tpo$ & {\textsc{(tso-fr)}} \\
      \hline
      $G.\fr(\tpo) ; G.\po$ 是非自反的 & {\textsc{(tso-coh)}} \\
      \hline
      $\domain(\tpo ; [G.V]) \subseteq G.V$ & {\textsc{(dsm-visible)}} \\
      \hline
      $\forall x \in \variables$. $\domain([\writeE_x \cup \RMW_x \cup \readE_x \cup \RexE_x]; \tpo; [G.P_x]) \subseteq G.P$ & {\textsc{(dsm-persist)}} \\
      \hline
      $\forall x \in \variables$. $\domain([\writeE_x \cup \RMW_x \cup \readE_x \cup \RexE_x]; \tpo ; [G.V \cap \flushE_x]) \subseteq G.P$ & {\textsc{(dsm-flush)}} \\
      \hline
    \end{tabular}
\end{frame}

\section{基于鲁棒性检查的验证方法}

\begin{frame}{正确性条件}
    \highlight{崩溃一致性}：对于可崩溃并发系统$S$的安全属性$\phi$，$\phi$在$S$的任意可达状态上恒成立。
    \begin{itemize}[<+->]
        \item \highlight{鲁棒性}是验证程序崩溃一致性的一种充分不必要条件
        \item 称程序是鲁棒的，当且仅当其任意可达状态均在无崩溃执行下可达
    \end{itemize}
\end{frame}

\begin{frame}{总体思路}
    检查鲁棒性要求遍历程序状态，并依次检查其可达性。例如，对于程序$a \assigns 1 \parallel b \assigns 1$

            给定状态$s$，如何检查$s$是否可达？

\end{frame}

\begin{frame}{恢复观测线程}
    由数学归纳法，只需检查给定状态是否在一次崩溃后可达。
    
    恢复观测线程可用于将状态嵌入到崩溃前的执行图中。

    \begin{columns}
    	\begin{column}{0.4\textwidth}
    		\begin{concurrentcode}
    			x \assigns 1; & y \assigns 2;  \\
    			\FLUSH x;     & \FLUSH y;       \\
    			a \assigns y; & b \assigns x;  \\
    			x \assigns a; & y \assigns b;  \\
    		\end{concurrentcode}
    	\end{column}
    
    	\begin{column}{0.5\textwidth}
    		\begin{figure}
    			\includegraphics[width=\linewidth]{pic/reduction-exp.pdf}
    		\end{figure}
    	\end{column}
    \end{columns}
\end{frame}

\begin{frame}{DTPO约束}
    使用恢复观测线程得到的执行图需额外满足一定条件。为此定义关于刷新指令的DTPO约束：
    \[
    G.\dtpo \triangleq \bigcup_{x \in \variables} \flushE_x \times \left\{ w \in \writeE_x \mid \exists w' \in \domain(G.\rrf), (w', w) \in G.\mo  \right\}
    \]
    \[
    G.\dtpo; \tpo~\text{是非自反的}
    \]

    \begin{block}{直观}
        刷新指令的效果是缓存中的值立刻写回主存。因此，假设恢复观测线程读到$w'$，则$w'$后发生的写事件$w$都一定发生在刷新指令之后，否则与恢复后观测到$w'$的事实矛盾。
    \end{block}
\end{frame}

\begin{frame}{可达性检查}
    \begin{block}{核心问题}
		\begin{center}
            给定状态$s$，如何检查$s$是否可达？
		\end{center}
	\end{block}

    \begin{itemize}
        \item 为程序插桩，引入恢复观测线程
        \item 得到插桩后的执行图
        \item 检查执行图是否满足公理以及DTPO约束
    \end{itemize}
\end{frame}

\begin{frame}{状态遍历}
    将约束编码为 SMT 公式，借助 DPLL 算法可高效遍历状态空间。

\begin{columns}
	\begin{column}{0.5\textwidth}
		\begin{align}
			\rho & = x_0 = 0 \land y_0 = 0 \tag{初始值}                   \\
			     & \land x_1 = 1 \land a = y_1 \land x_2 = a \tag{线程1}  \\
			     & \land y_2 = 2 \land b = x_3 \land y_3 = b \tag{线程2} \\
			     & \land r_1 = x_4 \land r_2 = y_4 \tag{恢复观测线程}
		\end{align}
	\end{column}

\section{基于不变式综合的验证方法}

\begin{frame}{正确性条件}
    \highlight{崩溃一致性}：对于可崩溃并发系统$S$的安全属性$\phi$，$\phi$在$S$的任意可达状态上恒成立。
    \begin{itemize}
        \item 类似于传统形式化验证中的安全属性检查
        \item 该类方法通常基于程序逻辑与不变式生成
    \end{itemize}
\end{frame}

\begin{frame}{程序逻辑}
	\begin{itemize}
		\item 串行程序：霍尔逻辑 $\{P\}~c~\{Q\}$
		\item 多线程并发程序：Owicki-Gries 推理$R, G \vdash \{P\}~c~\{Q\}$
		      \[
			      \inferrule{
				      R_1, G_1 \text{ 与 }~R_2, G_2 \text{ 不相干 } \\\\
				      R_1, G_1 \vdash \{P_1\}~c_1~\{Q_1\} \\\\
				      R_2, G_2 \vdash \{P_2\}~c_2~\{Q_2\}
			      }{
				      \{Q\} \cup R_1 \cup R_2, G_1 \cup G_2 \vdash \{P_1 \land P_2\}~c_1~\parallel~c_2~\{Q\}
			      }
		      \]
        \item 带崩溃的并发程序：？
	\end{itemize}
\end{frame}

\begin{frame}{模拟}
    程序逻辑无法直接对崩溃后的状态进行推理，故需引入\highlight{辅助变量}进行模拟。
    \vfill
    \begin{block}{辅助变量}
        \begin{enumerate}
            \item 对每个共享变量 $x \in \variables$，引入一个新的共享变量 $x'$，即其\highlight{副本}
            \item 程序状态同时为所有共享变量及其副本赋值
            \item $x'$的值无法直接操作，但是任意变量$x$ 都可以不确定地将值更新到$x'$
            \item $x$代表变量当前的缓存值，$x'$代表变量在主存中的实际值。崩溃后，从NVMM恢复的值是$x'$。
        \end{enumerate}
    \end{block}
    \vfill
    可以证明引入辅助变量后的内存模型和原始的模型语义等价。
\end{frame}

\begin{frame}{一种面向NVMM的程序逻辑（\S~5.3）}

    在Owicki-Gries推理的基础上可以得到面向 NVMM 的程序逻辑

    \[
    	\inferrule{
    		R_1, G_1 \text{ 与 }~R_2, G_2 \text{ 不相干 } \\\\
    		R_1, G_1 \vdash \{P_1\}~c_1~\{Q_1\} \\\\
    		R_2, G_2 \vdash \{P_2\}~c_2~\{Q_2\}
    	}{
    		\{Q\} \cup R_1 \cup R_2, G_1 \cup G_2 \vdash \{P_1 \land P_2\}~c_1~\parallel~c_2~\{Q\}
    	}
    \]

    \begin{definition}[不相干性]
    	$R_1, G_1$ 与 $R_2, G_2$ 不相干，当且仅当
    
    	\begin{enumerate}
    		\item $\forall (x, e, \mathbb{G}) \in G_1, \forall R \in R_2, R\land \mathbb{G}\rightarrow R[e/x]$
    		\item $\forall (x, e, \mathbb{G}) \in G_2, \forall R \in R_1, R\land \mathbb{G}\rightarrow R[e/x]$
    	\end{enumerate}
    	\label{def:non-interference}
    \end{definition}
\end{frame}

\begin{frame}{稳定性}
	证明树中的所有公式还应满足\highlight{稳定性}。稳定性约束无法用霍尔推导规则的形式给出。其在逻辑层面描述了共享变量非确定写回副本的过程。

	\vfill
	\begin{definition}[稳定性]
		若定义在 $\variables \cup \variables' \cup \registers$ 上的一阶逻辑公式 $\phi$ 满足
		\[
			\forall x \in \variables, \phi \rightarrow \phi[x/x']
		\]
		则称 $\phi$ 是\emph{稳定的}。
		\label{def:stability}
	\end{definition}
\end{frame}

\begin{frame}{示例}
    $\phi \triangleq z = 1 \rightarrow x = 1$

    \begin{figure}
    \begin{concurrentcode}
        $\{I_1: I\}$ & $\{I_5: I \land a = 1 \rightarrow y = 1\}$ \\
        x \assigns 1; & \WHILE a $\neq$ 1 \DO \\
        $\{I_2: I \land x = 1\}$ & \qquad $\{I_6: I\}$ \\
        \FLUSH x; & \qquad a \assigns y; \\
        $\{I_3: I \land x' = 1\}$ & \qquad $\{I_7: I \land a = 1 \rightarrow y = 1\}$ \\
        y \assigns 1; & $\{I_8: I \land y = 1\}$ \\
        $\{I_4: I\}$ & z \assigns 1; \\
        & $\{I_9: I\}$ \\
    \end{concurrentcode}
    \caption*{$I \triangleq x' = 1 \rightarrow x = 1 \land z' = 1 \rightarrow z = 1 \land y = 1 \rightarrow x' = 1 \land z = 1 \rightarrow x' = 1 \land z' = 1\rightarrow x' = 1$}
    \end{figure}
    
    称$I_1, \ldots, I_9$ 为对应程序位置的\highlight{不变式}。
\end{frame}

\begin{frame}{验证条件}

    为每个程序位置$i$引入未知谓词$I_i$，代表该位置的\emph{不变式}，则程序可被编码为验证条件。公式分为以下部分：
    \vfill
    \begin{itemize}
        \item 初始值
        \item 霍尔三元组有效性
        \item 不相干性
        \item 稳定性约束
        \item 安全属性
    \end{itemize}
\end{frame}

\begin{frame}{示例}
	\begin{align*}
		\psi & \triangleq x = y = z = a = x' = y' = z' = 0 \rightarrow I_1(x, y, z, a, x', y', z') \ldots \tag{初始化}                          \\
		     & \land I_1(x, y, z, a, x', y', z') \land x' = 1 \rightarrow I_2(x', y, z, a, x', y', z') \ldots \tag{有效性}                      \\
		     & \land I_1(x, y, z, a, x', y', z') \land I_8(x, y, z, a, x', y', z') \rightarrow I_1(x, y, 1, a, x', y', z') \ldots \tag{不相干性} \\
		     & \land I_1(x, y, z, a, x', y', z') \rightarrow  I_1(x, y, z, a, x, y', z') \ldots \tag{稳定性}                                    \\
		     & \land I_1(x, y, z, a, x', y', z') \rightarrow  \phi[\overrightarrow{x'/x}] \ldots \tag{安全性}                                   \\
	\end{align*}
\end{frame}

\begin{frame}{验证方法对比}
    \begin{itemize}[<+->]
        \item 两者都通过某种方式将崩溃消除，只考虑无崩溃执行
        \item 都需要额外的约束建模崩溃的语义
        \item 鲁棒性验证基于声明语义，而不变式综合基于操作语义
        \item 声明语义要求访存操作有限，故处理无界循环时有误报
    \end{itemize}
\end{frame}
