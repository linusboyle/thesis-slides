\begin{frame}{研究目的}

    非易失性内存的特性有别于传统存储设备，为编程带来新的挑战

    \vspace{1em}

    \begin{itemize}[<+->]
        \item NVM具有非易失性
        \begin{itemize}
            \item 程序的执行效果能够被保留到系统崩溃后……
            \item ……但这也包括程序中已经发生的错误 
        \end{itemize}
        \item 对内存的写操作是异步生效的
        \begin{itemize}
            \item 程序的执行效果可能只保留了一部分……
            \item ……并且它们是乱序写回的
        \end{itemize}
        \item 系统崩溃在不同场景下的作用范围不同
        \begin{itemize}
            \item 单机场景：完全崩溃模型
            \item 分布式场景：部分崩溃模型
        \end{itemize}
    \end{itemize}

\begin{frame}{正确性条件}

    无论采取何种语义和验证方式，NVM正确性保障的目标相同，即证明其\highlight{崩溃一致性}：
    \vfill
    \begin{block}{崩溃一致性}
        称运行于NVM的程序是崩溃一致的，当且仅当在其执行的任意时刻，其非易失性内存状态都满足给定的安全属性$\phi$。
    \end{block}
    \vfill
    直观理解：无论如何崩溃，程序恢复后的状态必然满足给定条件。
\end{frame}

\section[单机语义]{非易失性内存在完全崩溃模型下的形式化语义}

\begin{frame}{内存层级}
    单机场景下的NVM语义相对简单。以x86架构为例，影响语义的主要因素是\highlight{缓冲区}以及\highlight{崩溃}。
    \begin{figure}
        \includegraphics[width=0.6\linewidth]{pic/nvmm-hierarchy.pdf}
    \end{figure}
\end{frame}

\begin{frame}{缓冲区}
    % 该图独立存放在 tikz/buffer.tex，由 Makefile 预编译为 tikz/buffer.pdf。
    \centering
    \includegraphics{tikz/buffer.pdf}

    \vfill

    \begin{itemize}
        \item \highlight{写缓冲区}：线程局部所有，将写操作按顺序写回缓存 $\rightarrow$ 导致线程间的写-读重排
        \item \highlight{缓存}：所有线程共享，将同地址的写操作按顺序写回主存 $\rightarrow$ 导致对不同地址的写-写重排
        \item \highlight{NVM}：写操作到达此处即\highlight{持久化}，即使发生崩溃也不会丢失
    \end{itemize}
\end{frame}

\begin{frame}{崩溃}
    % 图独立存放在 tikz/buffer-crash.tex，由 Makefile 预编译。
    \centering
    \includegraphics{tikz/buffer-crash.pdf}

    \vfill

    \begin{itemize}
        \item 单机场景下的NVM采用\highlight{完全崩溃模型}
        \item 崩溃将导致所有数据丢失，除非其已持久化（到达NVM）
        \item 恢复后，程序从初始位置开始重新执行
    \end{itemize}
\end{frame}

\begin{frame}{声明语义：执行图}
    内存模型的声明语义是一组\highlight{执行图} $G$ 的集合。执行图为有向图，其节点是程序中的访存事件，带标签的边代表事件间的序关系
    \vspace{1em}
    \begin{itemize}
        \item 程序顺序$\po$ 刻画了指令被发射的顺序
        \item 读取顺序$\rf$ 刻画了读操作读到的值的来源
        \item 修改顺序$\mo$ 刻画了对同一地址的写操作的全局可见顺序
        \item ……

	\begin{block}{Px86 公理}

		存在$G.E \cap (\writeE \cup \RMW)$ 上的全序 $\nvo$，使得

		\begin{enumerate}
			\item $G.\hb = (G.\ppo \cup G.\rfe \cup G.\mo \cup G.\fr)^+$ 是非自反的
			\item $G.\fr ; G.\po$ 是非自反的
			\item $G.V = G.E$
			\item $G.\mo \subseteq \nvo$
			\item 如果$(e_1, e_2) \in G.\hb$ 且 $\type(e_1) \in \{\writeE, \RMW\}$, $\type(e_2) = \flushE$, $\location(e_1) = \location(e_2)$，则$e_1 \in G.P$

		\end{enumerate}
	\end{block}

\begin{frame}{声明语义：执行链}

    分布式场景下的NVM声明语义由执行图与崩溃的交错序列$G_1, \Xi_1, G_2, \Xi_2, \ldots, \Xi_{n-1}, G_n$刻画，称为\highlight{执行链}。

    \vfill
    
    \begin{itemize}
        \item 执行链中的每一张图表示一次无崩溃执行。
        \item 每张执行图部分继承上一张执行图的事件集，取决于崩溃发生的位置。
    \end{itemize} 
    \begin{align*}
    	\nabla_x(G, \Xi) & = \{(\bot, 0, \writeE(x, m(G, \Xi)(x)))\}                                                                      \\
    	                 & \cup \{(\theta(x) \setminus \Xi, 0, \writeE(x, \valw(\zeta(x)))) \mid \theta(x) \setminus \Xi \neq \emptyset\} \\
    	                 & \cup \{e \mid e \in E_x \setminus V_x \land \node(\src(e)) \not\in\Xi\}
    \end{align*}

    \vfill

    直观上讲：崩溃所影响的节点数据将丢失，除非其已到达具有NVM的内存节点，或者已被其它未崩溃的处理节点缓存

\begin{frame}{比较内存模型的强弱}

    \begin{center}
        NVM在不同场景下的语义差异是什么？
    \end{center}

    由于目前大量的相关研究所使用的内存模型不同，确定其强弱关系对于理解相关系统的兼容性至关重要。
    \vfill

    \begin{block}{精化}
        若对任意程序 $P$，在 $M_1$ 下可达的每个程序状态 $s \in Q_P$ 在 $M_2$ 下也可达，则称 $M_1$ 精化 $M_2$，记作 $M_1 \preccurlyeq M_2$。
    \end{block}

    \vfill
    如果$M_1 \preccurlyeq M_2$成立，则任何在 $M_2$ 下成立的安全属性在 $M_1$ 下也同样成立。因此，假设存在某系统基于$M_2$语义开发，那么在$M_1$下其正确性依然成立。
\end{frame}

\section[鲁棒性检验]{基于鲁棒性检验的形式化验证方法}

\begin{frame}{正确性条件}

    回顾崩溃一致性的定义：
    \vfill
    \begin{block}{崩溃一致性}
        称运行于NVM的程序是崩溃一致的，当且仅当在其执行的任意时刻，其非易失性内存状态都满足给定的安全属性$\phi$。
    \end{block}
    \vfill
    \begin{center}
        验证崩溃一致性要求用户给出规约，但这通常比较困难。
    \end{center}
\end{frame}

\begin{frame}{鲁棒性}
    \begin{itemize}
        \item \highlight{鲁棒性}是验证程序崩溃一致性的一种充分不必要条件
        \item 称程序是鲁棒的，当且仅当其任意可达状态均在无崩溃执行下可达
    \end{itemize}
    \vfill
    \centering
    % 图独立存放在 tikz/robustness.tex，由 Makefile 预编译。
    \includegraphics{tikz/robustness.pdf}
\end{frame}

\begin{frame}{总体思路}
    鲁棒性检验要求遍历可能的NVM状态，并依次检查是否可达；此外，需检查其是否在无崩溃情况下可达。
    
    \vfill
    例如，对于程序$a \assigns 1 \parallel b \assigns 1$

    \vfill
    \centering
    \begin{tabular}{ccc}
    \toprule
      程序状态 & 是否可达 & 是否无崩溃可达 \\ 
    \midrule
     $(a \mapsto 0, b \mapsto 0)$ & \checkmark  & \checkmark \\ 
     $(a \mapsto 0, b \mapsto 1)$ & \checkmark  & \checkmark \\ 
     $(a \mapsto 1, b \mapsto 0)$ & \checkmark  & \checkmark \\ 
     $(a \mapsto 1, b \mapsto 1)$ & \checkmark  & \checkmark \\ 
    \bottomrule
    \end{tabular}

	\begin{block}{核心问题}
		\begin{center}
            给定状态$s$，如何检查$s$是否可达？
		\end{center}
	\end{block}

\end{frame}

\begin{frame}{恢复观测线程}
    由数学归纳法，只需检查给定状态是否在一次崩溃后可达。
    \vfill
    恢复观测线程是在原本程序之外的一个\highlight{虚拟线程}，包含对所有内存位置的一次读取，作用是将NVM状态嵌入到崩溃前的执行图中。

    \begin{columns}
    	\begin{column}{0.4\textwidth}
    		\begin{concurrentcode}
    			x \assigns 1; & y \assigns 2;  \\
    			\FLUSH x;     & \FLUSH y;       \\
    			a \assigns y; & b \assigns x;  \\
    			x \assigns a; & y \assigns b;  \\
    		\end{concurrentcode}
    	\end{column}
    
    	\begin{column}{0.5\textwidth}
    		\begin{figure}
    			\includegraphics[width=\linewidth]{pic/reduction-exp.pdf}
    		\end{figure}
    	\end{column}
    \end{columns}
\end{frame}

\begin{frame}{DTPO约束}
    除前述的 NVM公理，带有恢复观测线程的执行图需额外满足一定条件，方能准确建模语义。
    
    \vfill
    为此定义关于刷新指令的DTPO约束：
    \[
    G.\dtpo \triangleq \bigcup_{x \in \variables} \flushE_x \times \left\{ w \in \writeE_x \mid \exists w' \in \domain(G.\rrf), (w', w) \in G.\mo  \right\}
    \]
    及额外公理：
    \[
    G.\dtpo; \tpo~\text{是非自反的}
    \]
\end{frame}

\begin{frame}{直观理解DTPO}

    \vspace{0.2em}
    \centering
    % 图独立存放在 tikz/dtpo.tex，由 Makefile 预编译。
    \includegraphics[scale=0.92]{tikz/dtpo.pdf}
    
    \begin{block}{直观}
        刷新指令的效果是缓存中的值立刻写回主存。因此，假设恢复观测线程读到$w'$，则$w'$后发生的写事件$w$都一定发生在刷新指令之后，否则与恢复后观测到$w'$的事实矛盾。
    \end{block}
\end{frame}

\begin{frame}{可达性检查}
    \begin{block}{核心问题}
		\begin{center}
            给定状态$s$，如何检查$s$是否可达？
		\end{center}
	\end{block}

    \begin{itemize}
        \item 为程序插桩，引入恢复观测线程
        \item 得到插桩后的执行图
        \item 检查执行图是否满足公理以及DTPO约束
    \end{itemize}
\end{frame}

\begin{frame}{状态遍历}
    将约束编码为 SMT 公式，借助 DPLL 算法可高效遍历状态空间。

\section[不变式综合]{基于不变式综合的形式化验证方法}

\begin{frame}{程序逻辑}
    另一种常见的验证方法是基于程序逻辑生成验证条件
	\begin{itemize}
		\item 串行程序：霍尔逻辑 $\{P\}~c~\{Q\}$
		\item 多线程并发程序：Owicki-Gries 推理$R, G \vdash \{P\}~c~\{Q\}$
        \begin{equation}
			      \inferrule{
				      R_1, G_1 \text{ 与 }~R_2, G_2 \text{ 不相干 } \\\\
				      R_1, G_1 \vdash \{P_1\}~c_1~\{Q_1\} \\\\
				      R_2, G_2 \vdash \{P_2\}~c_2~\{Q_2\}
			      }{
				      \{Q\} \cup R_1 \cup R_2, G_1 \cup G_2 \vdash \{P_1 \land P_2\}~c_1~\parallel~c_2~\{Q\}
			      }
                  \tag{不相干性}
        \end{equation}
        \item 运行于NVM 的并发程序：？
	\end{itemize}
\end{frame}

\begin{frame}{一种面向NVM的程序逻辑}

    在Owicki-Gries推理的基础上可以得到一种适用于 NVM 的程序逻辑。
    \vfill
    \highlight{主要挑战}：程序逻辑无法直接对崩溃后的状态进行推理。
    \vfill
    \begin{block}{辅助变量}
        \begin{enumerate}
            \item 对每个共享变量 $x$，引入一个新的共享变量 $x'$
            \item $x$代表变量当前在缓存中的值，$x'$则代表变量在主存中的实际值。
            \item 崩溃后，从NVM恢复的值是$x'$。
        \end{enumerate}
    \end{block}
    \vfill
    可以证明引入辅助变量后的内存模型和原始的模型语义等价。
\end{frame}

\begin{frame}{稳定性}
	除引入辅助变量，证明中的所有公式还需额外满足\highlight{稳定性}约束。

	\vfill
	\begin{definition}[稳定性]
		若定义在 $\variables \cup \variables' \cup \registers$ 上的一阶逻辑公式 $\phi$ 满足
		\[
			\forall x \in \variables, \phi \rightarrow \phi[x\backslash x']
		\]
		则称 $\phi$ 是\emph{稳定的}。
		\label{def:stability}
	\end{definition}
    \vfill
    稳定性约束无法用霍尔推导规则的形式给出。其在逻辑层面描述了共享变量非确定地写回副本的过程。
\end{frame}

\begin{frame}{示例}
    目标：证明$\phi \triangleq z = 1 \rightarrow x = 1$在非易失性内存状态上恒成立

    \begin{figure}
    \begin{concurrentcode}
        $\{I_1: I\}$ & $\{I_5: I \land a = 1 \rightarrow y = 1\}$ \\
        x \assigns 1; & \WHILE a $\neq$ 1 \DO \\
        $\{I_2: I \land x = 1\}$ & \qquad $\{I_6: I\}$ \\
        \FLUSH x; & \qquad a \assigns y; \\
        $\{I_3: I \land x' = 1\}$ & \qquad $\{I_7: I \land a = 1 \rightarrow y = 1\}$ \\
        y \assigns 1; & $\{I_8: I \land y = 1\}$ \\
        $\{I_4: I\}$ & z \assigns 1; \\
        & $\{I_9: I\}$ \\
    \end{concurrentcode}
    \caption*{$I \triangleq x' = 1 \rightarrow x = 1 \land z' = 1 \rightarrow z = 1 \land y = 1 \rightarrow x' = 1 \land z = 1 \rightarrow x' = 1 \land z' = 1\rightarrow x' = 1$}
    \end{figure}
    
    称$I_1, \ldots, I_9$ 为对应程序位置的\highlight{不变式}。
\end{frame}

\begin{frame}{验证条件}

    为每个程序位置$i$引入未知谓词$I_i$，代表该位置的\emph{不变式}，则程序可被编码为验证条件。公式分为以下部分：
    \vfill
    \begin{itemize}
        \item 初始值
        \item 霍尔三元组有效性
        \item 不相干性
        \item 稳定性约束
        \item 安全属性
    \end{itemize}
\end{frame}

\begin{frame}{VC编码示例}
	\begin{align*}
		\psi & \triangleq x = y = z = a = x' = y' = z' = 0 \rightarrow I_1(x, y, z, a, x', y', z') \ldots \tag{初始化}                          \\
		     & \land I_1(x, y, z, a, x', y', z') \land x' = 1 \rightarrow I_2(x', y, z, a, x', y', z') \ldots \tag{有效性}                      \\
		     & \land I_1(x, y, z, a, x', y', z') \land I_8(x, y, z, a, x', y', z') \rightarrow I_1(x, y, 1, a, x', y', z') \ldots \tag{不相干性} \\
		     & \land I_1(x, y, z, a, x', y', z') \rightarrow  I_1(x, y, z, a, x, y', z') \ldots \tag{稳定性}                                    \\
		     & \land I_1(x, y, z, a, x', y', z') \rightarrow  \phi[\overrightarrow{x'/x}] \ldots \tag{安全性}                                   \\
	\end{align*}
\end{frame}

\end{frame}

\begin{frame}{验证方法对比}
    \begin{itemize}[<+->]
        \item 两者都通过某种方式将崩溃消除，只考虑无崩溃执行
        \item 都需要额外的约束以建模崩溃：DTPO约束 v.s. 稳定性约束
        \item 鲁棒性验证基于声明语义，而不变式综合基于操作语义

    \end{itemize}
\end{frame}
