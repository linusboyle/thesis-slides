% 于是任一刷新事件 fl 与 w 之间都存在 dtpo 边（fl 在 w 之前）。

\begin{tikzpicture}[
    executiongraph,
    rrfarrow/.style={->, dashed, color=cyan!75!black,
                     line width=1pt, shorten >=2pt, shorten <=2pt},
    dtpoarrow/.style={->, dotted, color=brown,
                      line width=1.1pt, shorten >=2pt, shorten <=2pt}
]
    % 两个写事件与恢复观测读。
    \node[persisted] (wp) at (0,2.0) {$w'{:}\,\writeE(x, v')$};
    \node[visible] (w) at (3.9,2.0) {$w{:}\,\writeE(x, v)$};
    \node[visible] (r) at (0,3.9) {$\readE(x, v')$};

    % 恢复观测线程读到 w'。
    \draw[dashed, black!45, rounded corners=4pt] (-0.8,3.45) rectangle (0.8,4.35);
    \node[font=\scriptsize, text=black!60] at (0,4.5) {恢复观测线程};

    % 刷新事件。
    \node[visible] (fl) at (2.0,3.9) {$\flushE(x)$};

    \draw[rrfarrow] (wp.north) -- node[right, font=\scriptsize] {\rrf} (r.south);
    \draw[moarrow] (wp) -- node[above] {\mo} (w);
    \draw[dtpoarrow] (fl.south) -- node[above right, font=\scriptsize] {\dtpo} (w.north);
\end{tikzpicture}
